% Project Overview
\begin{frame}{Project Overview}
\textbf{Problem Statement:}
\begin{itemize}
\item Centralized EHR systems are vulnerable to breaches
\item Patients lack control over their medical records
\item Poor interoperability between healthcare providers
\end{itemize}

\vspace{0.3cm}
\textbf{MedLedger Solution:}
\begin{itemize}
\item Blockchain-based decentralized prescription management
\item End-to-end encrypted patient data
\item Dual-signature workflow (doctor + patient approval)
\item Patient-controlled access delegation
\end{itemize}
\end{frame}

\begin{frame}{Complete Workflow}
\textbf{Prescription Creation \& Access:}

\vspace{0.2cm}
\begin{center}
\textit{[Sequence diagram: Doctor → Backend → Patient → Blockchain]}
\end{center}

\vspace{0.2cm}
\textbf{Key Steps:}
\begin{enumerate}
\item Doctor encrypts prescription and signs (EIP-712)
\item Backend stores encrypted bundle on IPFS
\item Patient decrypts, reviews, and co-signs
\item Both signatures verified on-chain
\item Immutable record created
\item Patient can delegate access (automatic re-encryption)
\end{enumerate}

% Encryption Flow
\begin{frame}{Encryption Architecture}
\textbf{How Encryption Works:}

\vspace{0.2cm}
\begin{center}
\textit{[Diagram: Encryption flow with DEK and KEKs]}
\end{center}

\vspace{0.2cm}
\textbf{Encryption Process:}
\begin{enumerate}
\item Generate random DEK
\item Encrypt prescription data with DEK
\item For each recipient: derive KEK and encrypt DEK
\item Store encrypted bundle on IPFS
\end{enumerate}

\vspace{0.2cm}
\textbf{Decryption Process:}
\begin{enumerate}
\item User derives their KEK (signature or address-based)
\item Decrypt DEK using KEK
\item Decrypt prescription using DEK
\end{enumerate}
\end{frame}

% Feature 3: EIP-712
\begin{frame}{Feature 3: EIP-712 Dual-Signature}
\textbf{Secure Off-Chain Signing}

\vspace{0.2cm}
\textbf{What is EIP-712?}
\begin{itemize}
\item Standard for typed structured data signing
\item Human-readable signatures
\item Prevents phishing attacks
\end{itemize}

\vspace{0.2cm}
\textbf{Our Implementation:}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\textbf{Dual-Signature Flow:}
\begin{enumerate}
\item Doctor signs prescription data
\item Patient reviews and co-signs
\item Both signatures submitted on-chain
\item Contract verifies both signatures
\item Prescription registered if valid
\end{enumerate}
\end{column}

\begin{column}{0.48\textwidth}
\textbf{Security Features:}
\begin{itemize}
\item \textbf{Replay Protection:} Nonce tracking
\item \textbf{Expiration:} Time-limited signatures
\item \textbf{Domain Separation:} App-specific
\item \textbf{Signature Recovery:} ECDSA verification
\end{itemize}
\end{column}
\end{columns}
\end{frame}

% Automatic Re-encryption
\begin{frame}{Automatic Re-encryption}
\textbf{Seamless Delegate Access}

\vspace{0.2cm}
\textbf{Scenario:} Patient approves a new delegate (e.g., specialist doctor)

\vspace{0.2cm}
\textbf{Workflow:}
\begin{enumerate}
\item Patient sets delegate on-chain: \texttt{setDelegate(doctorAddress, true)}
\item System fetches all patient's prescriptions
\item For each encrypted prescription:
    \begin{itemize}
    \item Fetch encrypted bundle from IPFS
    \item Patient decrypts with their key
    \item Add delegate to recipients list (address-based key)
    \item Re-encrypt and upload to IPFS
    \item Update on-chain metadata URI
    \end{itemize}
\item Progress UI shows: "Re-encrypting 3/10..."
\end{enumerate}

\vspace{0.2cm}
\textbf{Benefits:}
\begin{itemize}
\item Fully automatic (no manual steps)
\item Parallel processing (faster)
\item Error handling (continues on failure)
\end{itemize}
\end{frame}

% Multi-Layer Security
\begin{frame}{Security Architecture}
\textbf{Defense in Depth - Five Layers:}

\vspace{0.2cm}
\begin{center}
\textit{[Diagram: 5 concentric security circles]}
\end{center}

\vspace{0.2cm}
\begin{enumerate}
\item \textbf{Blockchain Layer:} Immutability, decentralization, consensus
\item \textbf{Smart Contract Layer:} Access control, oracle verification, dual-signature
\item \textbf{Cryptography Layer:} AES-256-GCM, SHA-256, ECDSA
\item \textbf{Storage Layer:} IPFS content addressing, pinning
\item \textbf{Application Layer:} HTTPS, input validation, client-side encryption
\end{enumerate}

\vspace{0.2cm}
\textbf{Key Security Properties:}
\begin{itemize}
\item Backend never sees plaintext data
\item IPFS stores only encrypted ciphertext
\item Blockchain enforces access control
\end{itemize}
\end{frame}

% Challenges & Lessons
\begin{frame}{Key Challenges \& Lessons}
\textbf{Implementation Challenges:}
\begin{itemize}
\item \textbf{Key Management UX:} Solved with wallet signature-based derivation
\item \textbf{Re-encryption Performance:} Optimized with parallel processing
\item \textbf{IPFS Mutability:} Added \texttt{updatePrescriptionMetadata()} function
\item \textbf{Oracle Centralization:} Mitigated with transparent on-chain events
\end{itemize}

\vspace{0.3cm}
\textbf{Key Lessons Learned:}
\begin{itemize}
\item Encryption latency acceptable in healthcare context
\item Hybrid key derivation balances security and UX
\item Frontend caching essential for performance
\item Comprehensive testing catches bugs early
