% Introduction
\begin{frame}{Introduction}
\textbf{Healthcare Data Management Challenges:}
\begin{itemize}
\item Traditional Electronic Health Records (EHR) systems are centralized and vulnerable to single points of failure
\item Patient data ownership and control are limited
\item Interoperability between healthcare providers is lacking
\item Data breaches and unauthorized access are major concerns
\end{itemize}

\vspace{0.5cm}
\textbf{Significance of Blockchain in Healthcare:}
\begin{itemize}
\item \textbf{Decentralization:} Eliminates single point of failure and central authority control
\item \textbf{Immutability:} Creates tamper-proof audit trails for medical records
\item \textbf{Patient Sovereignty:} Patients control who accesses their health data
\item \textbf{Transparency:} All transactions are traceable and verifiable
\item \textbf{Interoperability:} Standardized data sharing across healthcare providers
\end{itemize}
\end{frame}

% Literature Review
\begin{frame}{Literature Review}
\textbf{Existing Blockchain-based Healthcare Solutions:}

\begin{enumerate}
\item \textbf{MedRec (Azaria et al., 2016)}
    \begin{itemize}
    \item Ethereum-based EHR management with smart contracts
    \item Limited scalability due to Layer-1 constraints
    \item No patient co-signature requirement
    \end{itemize}

\item \textbf{HealthChain (Husnain et al., 2024)}
    \begin{itemize}
    \item IPFS integration for off-chain storage
    \item Partial Layer-2 scaling solution
    \item Lacks draft-approval workflow
    \end{itemize}

\item \textbf{Block MedCare (Simonoski \& Bogatinoska, 2024)}
    \begin{itemize}
    \item Focus on interoperability standards
    \item Centralized components for metadata management
    \end{itemize}

\item \textbf{Shuaib et al. (2022)}
    \begin{itemize}
    \item Role-based access control implementation
    \item High gas costs on mainnet deployment
    \end{itemize}
\end{enumerate}
\end{frame}

% Motivation
\begin{frame}{Motivation}
\textbf{Research Gap Identified:}
\begin{itemize}
\item Existing solutions lack \textbf{patient co-signature} mechanisms---doctors can create records without patient consent
\item No \textbf{draft-approval workflow} for patients to review prescriptions before blockchain commitment
\item Limited use of \textbf{Layer-2 scaling} solutions to reduce transaction costs
\item Insufficient \textbf{flexible delegation} for granular access control
\end{itemize}

\vspace{0.5cm}
\textbf{Novelty of Our Approach:}
\begin{itemize}
\item \textbf{Dual-signature workflow:} Doctor drafts, patient finalizes---preventing unauthorized record creation
\item \textbf{Draft system:} Patients review before on-chain commitment
\item \textbf{Polygon Layer-2:} Scalable and cost-efficient transactions
\item \textbf{Patient-controlled delegation:} Granular access management via smart contracts
\item \textbf{IPFS integration:} Decentralized storage with on-chain metadata anchoring
\end{itemize}
\end{frame}

% Problem Statement
\begin{frame}{Problem Statement}
\begin{center}
\textbf{How can we design a decentralized health record system that ensures:}
\end{center}

\vspace{0.5cm}
\begin{enumerate}
\item \textbf{Patient Sovereignty:} Patients have full control over their medical records and who can access them

\item \textbf{Dual Authorization:} Both doctor and patient must approve prescriptions before they become immutable records

\item \textbf{Secure \& Transparent Access:} All access is logged on blockchain with role-based permissions

\item \textbf{Scalability:} System remains cost-effective and fast as the number of users grows

\item \textbf{Decentralized Storage:} Medical documents stored off-chain (IPFS) with cryptographic verification on-chain
\end{enumerate}

\vspace{0.3cm}
\begin{center}
\textbf{Challenge:} Balance decentralization, security, scalability, and patient control without compromising performance or usability.
\end{center}
\end{frame}

% System Architecture Implementation
\begin{frame}{System Architecture - Three-Layer Design}
\textbf{1. Blockchain Layer (Polygon Network)}
\begin{itemize}
\item Smart contract: PrescriptionRegistry.sol deployed and tested
\item Dual-signature workflow: Doctor drafts, patient finalizes
\item Role-based access control with delegation support
\item Event emission for transparency
\end{itemize}

\textbf{2. Storage Layer (IPFS via Pinata)}
\begin{itemize}
\item Off-chain encrypted document storage
\item Content-addressed with cryptographic hashes
\item Metadata anchored on blockchain
\end{itemize}

\textbf{3. Application Layer}
\begin{itemize}
\item React.js frontend with Tailwind CSS
\item MetaMask wallet integration via Wagmi
\item Node.js/Express backend API
\item Real-time metrics collection
\end{itemize}
\end{frame}

% Smart Contract Details
\begin{frame}[fragile]{Smart Contract Implementation}
\textbf{PrescriptionRegistry.sol - Core Functions:}

\begin{itemize}
\item \texttt{submitDraft(address patient)} - Doctor creates prescription draft on-chain
\item \texttt{finalizeDraft(uint256 draftId, string metadataURI)} - Patient co-signs and publishes
\item \texttt{setDelegate(address viewer, bool allowed)} - Patient grants/revokes access
\item \texttt{getPrescription(uint256 prescriptionId)} - Retrieve prescription metadata
\item \texttt{canView(uint256 prescriptionId, address viewer)} - Check authorization
\end{itemize}

\vspace{0.3cm}
\textbf{Key Features:}
\begin{itemize}
\item Allow-listed doctor addresses for security
\item Patient-controlled access via delegation
\item Immutable audit trail with events
\item Gas-efficient design
\end{itemize}
\end{frame}

% Workflow Implementation
\begin{frame}{Implemented Workflow}
\textbf{Prescription Creation \& Approval Process:}

\begin{enumerate}
\item \textbf{Doctor Draft Submission}
    \begin{itemize}
    \item Doctor calls \texttt{submitDraft(patient)} on smart contract
    \item Receives \texttt{draftId} from blockchain
    \item Uploads metadata to Pinata (IPFS)
    \item Backend stores draft in request queue
    \end{itemize}

\item \textbf{Patient Review \& Approval}
    \begin{itemize}
    \item Patient fetches pending drafts via API
    \item Reviews prescription details
    \item Calls \texttt{finalizeDraft(draftId, metadataURI)} from wallet
    \item Smart contract mints immutable prescription record
    \end{itemize}

\item \textbf{Access Control}
    \begin{itemize}
    \item Patient grants delegation via \texttt{setDelegate(doctor, true)}
    \item Authorized viewers retrieve prescriptions
    \item All access logged on blockchain
    \end{itemize}
\end{enumerate}
\end{frame}

% Security Analysis
\begin{frame}{Security Implementation}
\textbf{Multi-Layer Security Approach:}

\textbf{1. Smart Contract Level:}
\begin{itemize}
\item Allow-list for doctor addresses (prevents unauthorized drafts)
\item Patient-only finalization (no unauthorized publishing)
\item On-chain delegation verification (trustless access control)
\item Custom error messages for security violations
\end{itemize}

\textbf{2. API Level:}
\begin{itemize}
\item Address validation with regex patterns
\item Sender verification via \texttt{x-sender} header
\item Transaction hash validation (64-character hex)
\item Role-based endpoint access control
\end{itemize}

\textbf{3. Storage Level:}
\begin{itemize}
\item IPFS content addressing (tamper-proof)
\item Metadata URI stored on-chain (immutable reference)
\item Planned: AES encryption for sensitive data
\end{itemize}
\end{frame}

% Limitations
\begin{frame}{Current Limitations}
\textbf{Identified Limitations:}

\begin{enumerate}
\item \textbf{Scalability:}
    \begin{itemize}
    \item JSON file-based storage not suitable for production
    \item Need database migration (PostgreSQL/MongoDB)
    \end{itemize}

\item \textbf{Security:}
    \begin{itemize}
    \item x-sender header trust-based (requires EIP-712 signatures)
    \item No encryption for IPFS-stored data yet
    \end{itemize}

\item \textbf{Functionality:}
    \begin{itemize}
    \item No emergency access override mechanism
    \item Limited to prescriptions (lab reports planned)
    \item Manual refresh required (no real-time notifications)
    \end{itemize}

\item \textbf{Compliance:}
    \begin{itemize}
    \item Need HIPAA/GDPR compliance audit
    \item Regulatory framework integration pending
    \end{itemize}
\end{enumerate}
